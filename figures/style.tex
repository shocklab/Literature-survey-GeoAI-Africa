% =====================================================================
%  Base axis style: no box, no chart junk, grid only on the value axis.
% =====================================================================
\pgfplotsset{
  jsbase/.style={
    axis line style={draw=jsAxis, line width=0.5pt},
    tick label style={font=\footnotesize, /pgf/number format/assume math mode=true},
    label style={font=\footnotesize},
    title style={font=\small\bfseries, at={(0,1)}, anchor=south west,
                 yshift=4pt, xshift=-2pt},
    legend style={font=\footnotesize, draw=none, fill=none,
                  cells={anchor=west}, /tikz/every even column/.append style={column sep=6pt}},
    tick align=outside,
    tickwidth=2.5pt,
    enlarge x limits=0.04,
    enlarge y limits=0.04,
    clip=false,
  },
  % vertical bars, horizontal gridlines
  jsybar/.style={
    jsbase, ybar,
    axis x line*=bottom, axis y line*=left,
    ymajorgrids, grid style={draw=jsGrid, line width=0.4pt},
    axis on top=false,
    ymin=0,
    bar width=11pt,
    every node near coord/.append style={font=\scriptsize, color=jsAxis,
        /pgf/number format/assume math mode=true},
    nodes near coords align={vertical},
  },
  % horizontal bars, vertical gridlines
  jsxbar/.style={
    jsbase, xbar,
    axis y line*=left, axis x line*=bottom,
    xmajorgrids, grid style={draw=jsGrid, line width=0.4pt},
    xmin=0,
    bar width=8pt,
    ytick style={draw=none},
    y tick label style={font=\footnotesize},
    every node near coord/.append style={font=\scriptsize, color=jsAxis, xshift=1pt,
        /pgf/number format/assume math mode=true},
    nodes near coords align={horizontal},
  },
  % ---- stacked variants: must NOT inherit plain ybar/xbar, which fight
  %      with "ybar stacked"/"xbar stacked" over bar shift.
  jsybarstack/.style={
    jsbase, ybar stacked,
    axis x line*=bottom, axis y line*=left,
    ymajorgrids, grid style={draw=jsGrid, line width=0.4pt},
    ymin=0, bar width=15pt,
  },
  jsxbarstack/.style={
    jsbase, xbar stacked,
    axis y line*=left, axis x line*=bottom,
    xmajorgrids, grid style={draw=jsGrid, line width=0.4pt},
    xmin=0, bar width=9pt,
    ytick style={draw=none},
    y tick label style={font=\footnotesize},
  },
}
